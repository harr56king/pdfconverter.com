    <script src="https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js"></script>
    <script>
        const { jsPDF } = window.jspdf;
        let images = [];

        // Drag & Drop Handlers
        const dropZone = document.getElementById('dropZone');
        const fileInput = document.getElementById('fileInput');

        dropZone.addEventListener('click', () => fileInput.click());
        dropZone.addEventListener('dragover', (e) => {
            e.preventDefault();
            dropZone.style.backgroundColor = '#f8f9fa';
        });
        dropZone.addEventListener('dragleave', () => {
            dropZone.style.backgroundColor = 'white';
        });
        dropZone.addEventListener('drop', (e) => {
            e.preventDefault();
            handleFiles(e.dataTransfer.files);
        });

        fileInput.addEventListener('change', () => handleFiles(fileInput.files));

        function handleFiles(files) {
            for (const file of files) {
                if (file.type.startsWith('image/')) {
                    const reader = new FileReader();
                    reader.onload = (e) => {
                        images.push({
                            data: e.target.result,
                            name: file.name
                        });
                        updatePreview();
                    };
                    reader.readAsDataURL(file);
                }
            }
        }

        function updatePreview() {
            const container = document.getElementById('previewContainer');
            container.innerHTML = images.map((img, index) => `
                <div class="preview-item">
                    <img src="${img.data}" alt="${img.name}">
                    <div class="image-actions">
                        <button onclick="moveImage(${index}, -1)"><i class="bi bi-arrow-up"></i></button>
                        <button onclick="moveImage(${index}, 1)"><i class="bi bi-arrow-down"></i></button>
                        <button onclick="removeImage(${index})"><i class="bi bi-x"></i></button>
                    </div>
                </div>
            `).join('');
        }

        function moveImage(index, direction) {
            if (index + direction >= 0 && index + direction < images.length) {
                [images[index], images[index + direction]] = [images[index + direction], images[index]];
                updatePreview();
            }
        }

        function removeImage(index) {
            images.splice(index, 1);
            updatePreview();
        }

        // PDF Generation
        async function generatePDF() {
            if (images.length === 0) return alert('Please add images first!');

            const doc = new jsPDF({
                orientation: document.getElementById('pageOrientation').value,
                unit: 'mm',
                format: document.getElementById('pageSize').value
            });

            const margin = parseInt(document.getElementById('marginSize').value);
            const borderColor = document.getElementById('borderColor').value;
            const pageSpread = document.getElementById('pageSpread').value;

            for (let i = 0; i < images.length; i++) {
                if (i !== 0) doc.addPage();
                
                const img = new Image();
                img.src = images[i].data;
                await new Promise(resolve => img.onload = resolve);

                const pageWidth = doc.internal.pageSize.getWidth();
                const pageHeight = doc.internal.pageSize.getHeight();
                
                const contentWidth = pageWidth - (margin * 2);
                const contentHeight = pageHeight - (margin * 2);
                
                // Draw border
                if (margin > 0) {
                    doc.setDrawColor(borderColor);
                    doc.rect(margin, margin, contentWidth, contentHeight);
                }

                // Calculate image dimensions
                const ratio = Math.min(contentWidth / img.width, contentHeight / img.height);
                const width = img.width * ratio;
                const height = img.height * ratio;
                const x = margin + (contentWidth - width) / 2;
                const y = margin + (contentHeight - height) / 2;

                doc.addImage(img, 'JPEG', x, y, width, height);
            }

            doc.save('converted.pdf');
        }

        function clearAll() {
            images = [];
            updatePreview();
        }

        // Update margin value display
        document.getElementById('marginSize').addEventListener('input', (e) => {
            document.getElementById('marginValue').textContent = e.target.value;
        });
    </script>
